Sync any data source into Notion.

Continuously upsert external records into a Notion Database with Workers, a declarative schema, and a persistent cursor.

zendeskSync.ts See example ↗ 
import { Worker } from "@notionhq/workers";
import { zendesk } from "@notionhq/integrations";

export const worker = Worker.sync({
  name: "ticketsSync",
  source: zendesk,
  destination: notion,
  transform(ticket) {
    return {
      title:    ticket.subject,
      status:   ticket.status,
      priority: ticket.priority,
      assignee: ticket.assignee_id,
    };
  },
});

Build any tool for your agents.

Write custom tools for Notion Agents that generate assets, query live data, and hit any API.

presentationTool.ts 
import { Tool } from "@notionhq/agents";

export const generatePresentation = Tool({
  name: "generate_presentation",
  description: "Create slides from a Notion page",
  parameters: {
    pageId:  { type: "string" },
    format:  { type: "string", enum: ["pptx", "pdf"] },
  },
  execute: async ({ pageId, format }) => {
    const page   = await notion.pages.retrieve({ page_id: pageId });
    const slides = await slides.generate({ page, format });
    return { url: slides.downloadUrl };
  },
});

import { Tool } from "@notionhq/agents";

export const queryRevenue = Tool({
  name: "query_revenue",
  description: "Query live revenue from the data warehouse",
  parameters: {
    startDate: { type: "string" },
    endDate:   { type: "string" },
    segment:   { type: "string" },
  },
  execute: async (params) => {
    const rows = await warehouse.query(
      `SELECT sum(mrr) FROM revenue
       WHERE date BETWEEN $1 AND $2
         AND segment = $3`,
      [params.startDate, params.endDate, params.segment]
    );
    return rows[0];
  },
});
